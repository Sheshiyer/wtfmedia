   export type TimelineSystem = 'uncut' | 'published';
   export type IntervalStatus = 'matched' | 'cut_from_published' | 'added_in_published' | 'conflicted';

   export interface AlignmentInterval {
     intervalIndex: number;

     uncutEndSec: number;
     pubStartSec: number;
     pubEndSec: number;
     status: IntervalStatus;
     confidence: number;
   }

   export interface TimelineAlignmentRecord {
     id: string;
     episodeId: string;
     uncutAssetId: string;
     publishedAssetId: string;
     algorithm: AlignmentAlgorithm;
     confidenceScore: number;
     status: AlignmentStatus;
     intervals: AlignmentInterval[];
     createdAt: string;
     updatedAt: string;
   }

   export interface CoordinateConversionResult {
     targetTimeSec: number | null;
     targetSystem: TimelineSystem;
     status: 'verified' | 'cut_from_published' | 'added_in_published' | 'unmapped' | 'conflicted';
     confidence: number;
     intervalIndex?: number;
     reason?: string;
   }

2. Create `cloudflare/src/timeline/alignment-engine.ts`:
   - Implement `class TimelineAlignmentEngine`:
     - Method `loadAlignment(db: D1Database, episodeId: string): Promise<TimelineAlignmentRecord | null>`.
     - Method `convertCoordinate(alignment: TimelineAlignmentRecord, fromSystem: TimelineSystem, timeSec: number): CoordinateConversionResult`.
   - **Algorithm**:
     - Binary search (`O(\log K)`) on sorted interval array:
       - If `fromSystem === 'uncut'`: find interval where `uncutStartSec <= timeSec <= uncutEndSec`.
       - If `fromSystem === 'published'`: find interval where `pubStartSec <= timeSec <= pubEndSec`.
     - If found:
       - If `interval.status === 'matched'`:
         - If `fromSystem === 'uncut'`:
           $$\text{targetTime} = \text{pubStartSec} + \frac{\text{pubEndSec} - \text{pubStartSec}}{\text{uncutEndSec} - \text{uncutStartSec}} \cdot (\text{timeSec} - \text{uncutStartSec})$$
         - If `fromSystem === 'published'`:
           $$\text{targetTime} = \text{uncutStartSec} + \frac{\text{uncutEndSec} - \text{uncutStartSec}}{\text{pubEndSec} - \text{pubStartSec}} \cdot (\text{timeSec} - \text{pubStartSec})$$
         - Return `{ targetTimeSec: targetTime, targetSystem, status: 'verified', confidence: interval.confidence, intervalIndex: interval.intervalIndex }`.
       - If `fromSystem === 'uncut'` and `interval.status === 'cut_from_published'`:
         - Return `{ targetTimeSec: null, targetSystem: 'published', status: 'cut_from_published', confidence: 1.0, reason: 'Moment was trimmed/excluded from the published YouTube cut' }`.
       - If `fromSystem === 'published'` and `interval.status === 'added_in_published'`:
         - Return `{ targetTimeSec: null, targetSystem: 'uncut', status: 'added_in_published', confidence: 1.0, reason: 'Intro bumper/sponsor graphics added in post-production, not in uncut studio recording' }`.
     - If no interval contains `timeSec`:
       - Return `{ targetTimeSec: null, targetSystem, status: 'unmapped', confidence: 0.0, reason: 'Timestamp falls outside mapped episode duration' }`.
